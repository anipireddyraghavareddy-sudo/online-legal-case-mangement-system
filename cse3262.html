// Status formatting function statusClass(status) { return status.toLowerCase().replace(/\s+/g, "-"); } function statusHTML(status) { return ` ${escapeHTML(status)} `; } // Format date function formatDate(dateString) { if (!dateString) return "Not scheduled"; const date = new Date(dateString + "T00:00:00"); if (isNaN(date.getTime())) return "Invalid date"; return date.toLocaleDateString("en-IN", { day: "2-digit", month: "short", year: "numeric" }); } // Basic HTML escaping for text displayed in the page function escapeHTML(value) { return String(value ?? "").replace(/[&<>"']/g, char => { const map = { "&": "&", "<": "<", ">": ">", '"': """, "'": "'" }; return map[char]; }); } // Dashboard statistics function renderStats() { const total = cases.length; const pending = cases.filter(item => item.status === "Pending" || item.status === "In Progress" ).length; const completed = cases.filter(item => item.status === "Completed" ).length; const today = makeDate(0); const upcoming = cases.filter(item => item.hearing >= today && item.status !== "Completed" ).length; document.getElementById("totalCases").textContent = total; document.getElementById("pendingCases").textContent = pending; document.getElementById("completedCases").textContent = completed; document.getElementById("upcomingCases").textContent = upcoming; } // Recent cases function renderRecent() { const recentTable = document.getElementById("recentTable"); const recent = cases.slice(0, 5); if (recent.length === 0) { recentTable.innerHTML = ` No cases available. `; return; } recentTable.innerHTML = recent.map(item => ` ${escapeHTML(item.id)} ${escapeHTML(item.title)} ${escapeHTML(item.client)} ${escapeHTML(item.lawyer)} ${formatDate(item.hearing)} ${statusHTML(item.status)} `).join(""); } // All cases function renderCases() { const table = document.getElementById("casesTable"); const search = searchInput.value.toLowerCase().trim(); const selectedStatus = statusFilter.value; const filtered = cases.filter(item => { const matchesSearch = item.title.toLowerCase().includes(search) || item.client.toLowerCase().includes(search) || item.lawyer.toLowerCase().includes(search) || item.id.toLowerCase().includes(search) || item.type.toLowerCase().includes(search); const matchesStatus = selectedStatus === "All" || item.status === selectedStatus; return matchesSearch && matchesStatus; }); if (filtered.length === 0) { table.innerHTML = ` No matching cases found. `; return; } table.innerHTML = filtered.map(item => ` ${escapeHTML(item.id)} ${escapeHTML(item.title)} ${escapeHTML(item.client)} ${escapeHTML(item.lawyer)} ${formatDate(item.hearing)} ${statusHTML(item.status)} `).join(""); table.querySelectorAll("[data-delete-id]").forEach(button => { button.addEventListener("click", () => { deleteCase(button.dataset.deleteId); }); }); } // Hearing schedule function renderHearings() { const list = document.getElementById("hearingList"); const today = makeDate(0); const upcoming = cases .filter(item => item.hearing >= today && item.status !== "Completed" ) .sort((a, b) => a.hearing.localeCompare(b.hearing)); if (upcoming.length === 0) { list.innerHTML = `

No upcoming hearings.

`; return; } list.innerHTML = upcoming.map(item => `

${escapeHTML(item.title)}

Client: ${escapeHTML(item.client)}

Lawyer: ${escapeHTML(item.lawyer)}

${escapeHTML(item.type)} Case

${formatDate(item.hearing)}
`).join(""); } // Client directory function renderClients() { const list = document.getElementById("clientList"); // Group cases by client name const clients = {}; cases.forEach(item => { if (!clients[item.client]) { clients[item.client] = []; } clients[item.client].push(item); }); const clientNames = Object.keys(clients); if (clientNames.length === 0) { list.innerHTML = `

No clients registered.

`; return; } list.innerHTML = clientNames.map(name => { const clientCases = clients[name]; return `

${escapeHTML(name)}

${clientCases.length} case(s) registered

Lawyer: ${escapeHTML(clientCases[0].lawyer)}

${clientCases.map(item => statusHTML(item.status) ).join(" ")}
`; }).join(""); } // Refresh everything function renderAll() { renderStats(); renderRecent(); renderCases(); renderHearings(); renderClients(); } // Initial load renderAll();
